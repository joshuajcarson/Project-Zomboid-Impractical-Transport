module Base
{
    /* the .X mesh is already in game units (Y up, stick ~0.83 units tall); change `scale` only to resize */
    model PogoStick_3d
    {
        mesh = ImpracticalTransport/PogoStick,
        texture = ImpracticalTransport/PogoStick_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            /* The held-item bone (Bip01_Prop1) is rolled 16.8 degrees about the rider's forward axis in the pose (measured in Blender, confirmed in game 2026-11-01 at the screen facings): R = Ry(16.8) Rx(90), written as rotate 90 0 -16.8. Pitch 90 stands the stick upright (the old 102 leaned it back). The offset cancels the grip point (0.69 up the stick) after the turn, then y is +0.1 to carry the stick slightly in front of the rider (+y is forward; 0.2 was too far for the hands). */
            rotate = 90.0000 0.0000 -16.8000,
            offset = -0.0989 0.1000 -0.6909,
        }
    }

    /* the tricycle and unicycle: R = Ry(16.8) Rx(90) Ry(180) written as rotate -90 0 -163.2 (see the pogo note); the offset cancels the grip point */
    model Tricycle_3d
    {
        mesh = ImpracticalTransport/Tricycle,
        texture = ImpracticalTransport/Tricycle_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            rotate = -90.0000 0.0000 -163.2000,
            offset = 0.0206 -0.2250 -0.5546,
        }
    }

    model Unicycle_3d
    {
        mesh = ImpracticalTransport/Unicycle,
        texture = ImpracticalTransport/Unicycle_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            rotate = -90.0000 0.0000 -163.2000,
            offset = -0.1734 -0.1050 -0.5744,
        }
    }
}
