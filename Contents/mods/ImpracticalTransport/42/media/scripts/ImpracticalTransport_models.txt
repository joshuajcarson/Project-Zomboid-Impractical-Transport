module Base
{
    /* the .X mesh is already in game units (Y up, stick ~0.55 units tall); change `scale` only to resize */
    model PogoStick_3d
    {
        mesh = ImpracticalTransport/PogoStick,
        texture = ImpracticalTransport/PogoStick_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            /* hand bone axes in the idle pose: X right, Y forward, Z up. Rotating 90 about X turns the stick's up into the hand's up so the foot hangs down. */
            rotate = 90.0000 0.0000 0.0000,
            /* the grips are 0.46 up the stick, which after that turn is 0.46 along the hand's Z: cancel it so the hand holds the grips */
            offset = 0.0000 0.0000 -0.4600,
        }
    }
}
