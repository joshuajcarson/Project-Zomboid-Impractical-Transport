module Base
{
    /* the .X mesh is already in game units (Y up, stick ~0.83 units tall); change `scale` only to resize */
    model PogoStick_3d
    {
        mesh = ImpracticalTransport/PogoStick,
        texture = ImpracticalTransport/PogoStick_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            /* hand bone axes in the idle pose: X right, Y forward, Z up. 78 about X = 90 (the stick's up becomes the hand's up) minus the 12 degree forward lean the rider has in Bob_PogoHop (LEAN_DEG in tools/blender/pogo_hop.py); 102 would lean it back. */
            rotate = 102.0000 0.0000 0.0000,
            /* the grip held is the stick's one at model x = -0.105 (0.75 game units per metre; flipped after a test showed the right hand on the other one), 0.69 up the stick. After the turn that point is at (-0.105, 0.69 cos 78, 0.69 sin 78): cancel it so the hand holds the grip. */
            offset = 0.1050 0.1434 -0.6749,
        }
    }

    /* the tricycle: attachment rotate and offset are fitted by screenshot (tools/make_trike_calibration.py); rotate x 90 stands the model up, y 180 turns it to face the rider's front */
    model Tricycle_3d
    {
        mesh = ImpracticalTransport/Tricycle,
        texture = ImpracticalTransport/Tricycle_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            rotate = 90.0000 180.0000 0.0000,
            offset = 0.1800 -0.2250 -0.5250,
        }
    }

    model Unicycle_3d
    {
        mesh = ImpracticalTransport/Unicycle,
        texture = ImpracticalTransport/Unicycle_texture,
        scale = 1,

        attachment Bip01_Prop1
        {
            rotate = 90.0000 180.0000 0.0000,
            offset = 0.0000 -0.1050 -0.6000,
        }
    }
}
